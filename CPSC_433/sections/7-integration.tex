\subsection{Algorithm 4: Optional combined pipeline}
\begin{algorithm}[H]
\caption{Combined Three-Paradigm Pipeline}
\begin{algorithmic}[1]
\Require Scheduling instance $I$
\Ensure Best valid timetable found, or FAILURE
\State $a_0\gets\Call{ORFeasibility}{I}$
\If{$a_0=\mathsf{FAILURE}$}
  \State \Return FAILURE
\EndIf
\State $a_1\gets\Call{SetBasedImprove}{I,a_0}$
\State $a_2\gets\Call{ANDOptimize}{I,a_1}$
\State \Return $a_2$ \Comment{Optimal only if AND stage completed}
\end{algorithmic}
\end{algorithm}
