\section{Model I: OR-Tree Feasibility Search}
\subsection{Search model: $A_{\lor}=(S_{\lor},T_{\lor})$}
\paragraph{Problem descriptions.}
\begin{equation}
\boxed{\Prob_{\lor}=\{(a,U):a\text{ is partial},\ U=X\setminus\domn(a)\}.}
\end{equation}
Here $a$ is the current partial timetable, and $U$ contains unscheduled components. A node has the lecture's form
\begin{equation}
  ((a,U),sol),\qquad sol\in\{\yes,\und,\no\}.
\end{equation}
\paragraph{Alternatives.}
Select a component $x\in U$ and define
\begin{equation}
  D_a(x)=\{s\in D(x):\Consistent(a\cup\{x\mapsto s\})\}.
\end{equation}
For every $s_i\in D_a(x)$, let
\begin{equation}
  pr_i=(a\cup\{x\mapsto s_i\},U\setminus\{x\}).
\end{equation}
Then the application-specific alternative relation is
\begin{equation}
  \boxed{\Altern(pr,pr_1,\ldots,pr_r),\qquad r=|D_a(x)|.}
\end{equation}
\paragraph{States and transitions.}
Following Lecture 11,
\begin{equation}
  S_{\lor}\subseteq\Otree,\qquad
  T_{\lor}=\{(q,q'): \Erw_{\lor}(q,q')\text{ or }\Erw_{\lor}^{*}(q,q')\}.
\end{equation}
The extension relation marks a solved leaf $\yes$, marks an unsolvable leaf $\no$, or expands a leaf using $\Altern$. Extensions can act on leaves inside a larger tree.

\subsection{Algorithm 1: OR-tree feasibility}
\begin{algorithm}[H]
\caption{OR-Tree Feasibility Search (operational outline)}
\begin{algorithmic}[1]
\Require Components $X$, domains $D$, partial checker $\Consistent$
\Ensure Complete valid timetable, or FAILURE
\State Initialize tree $q\gets((\varnothing,X),\und)$
\While{an unresolved leaf exists}
  \State $v\gets$ deepest unresolved leaf of $q$
  \State $(a,U)\gets$ problem description at $v$
  \If{$U=\varnothing$}
    \If{$\Const(a)$}
      \State Mark $v$ as $\yes$; \Return $a$
    \Else
      \State Mark $v$ as $\no$; \State \textbf{continue}
    \EndIf
  \EndIf
  \State $x\gets\arg\min_{y\in U}|D_a(y)|$ \Comment{MRV}
  \State $C\gets D_a(x)$
  \If{$C=\varnothing$}
    \State Mark $v$ as $\no$
  \Else
    \ForAll{$s\in C$ in deterministic slot order}
      \State Add child $((a\cup\{x\mapsto s\},U\setminus\{x\}),\und)$ to $v$
    \EndFor
  \EndIf
\EndWhile
\State \Return FAILURE
\end{algorithmic}
\end{algorithm}

\section{Model II: Set-Based Local Optimization}
\subsection{Search model: $A_{set}=(S_{set},T_{set})$}
\paragraph{Facts.}
\begin{equation}
  \boxed{F=\{\Assigned(x,s):x\in X,\ s\in D(x)\}.}
\end{equation}
The fact $\Assigned(x,s)$ means that component $x$ occupies weekly slot $s$.
\paragraph{States.}
\begin{equation}
  \boxed{S_{set}=\{q\subseteq F:q\text{ encodes one complete valid timetable}\}\subseteq2^F.}
\end{equation}
For any $q\in S_{set}$, let $a_q$ be the complete timetable that its facts represent.
\paragraph{Extensions.}
\begin{equation}
\begin{aligned}
\text{Move: }&\{\Assigned(x,s_1)\}\longrightarrow\{\Assigned(x,s_2)\};\\
\text{Swap: }&\{\Assigned(x,s_1),\Assigned(y,s_2)\}\\
&\hspace{13mm}\longrightarrow\{\Assigned(x,s_2),\Assigned(y,s_1)\}.
\end{aligned}
\end{equation}
Here $x,y$ are distinct components and $s_1,s_2$ are their current slots. The rules are applicable only if domains, preassignments, and all hard constraints are respected in the resulting timetable.
\begin{equation}
  \boxed{q'=(q\setminus A)\cup B,\qquad A\to B\in\EXT.}
\end{equation}
\begin{equation}
  T_{set}=\{(q,q'):A\to B\in\EXT,\ A\subseteq q,\ q'=(q\setminus A)\cup B,\ q'\in S_{set}\}.
\end{equation}

\subsection{Search process: $P_{set}=(A_{set},\Env_{set},K_{set})$}
The proposed greedy weight function is
\begin{equation}
  \boxed{f_{weight}(A,B,e)=\Eval(a_{(q\setminus A)\cup B}),}
\end{equation}
where the environment $e$ provides the current timetable $q$ and other problem information. The function $f_{select}$ breaks ties between equally weighted extensions.

\subsection{Algorithm 2: Set-based hill climbing}
\begin{algorithm}[H]
\caption{Set-Based Optimization}
\begin{algorithmic}[1]
\Require Feasible timetable $a_{\mathrm{feasible}}$, $\EXT$, $\Eval$
\Ensure Improved valid timetable
\State $q\gets\{\Assigned(x,a_{\mathrm{feasible}}(x)):x\in X\}$
\While{$\Eval(a_q)>0$}
  \State $N\gets\{q':(q,q')\in T_{set},\ q'\ne q\}$
  \If{$N=\varnothing$}
    \State \Return $a_q$
  \EndIf
  \State $m\gets\min_{q'\in N}\Eval(a_{q'})$
  \If{$m\ge\Eval(a_q)$}
    \State \Return $a_q$ \Comment{Local optimum}
  \EndIf
  \State $q\gets f_{select}(\{q'\in N:\Eval(a_{q'})=m\})$
\EndWhile
\State \Return $a_q$
\end{algorithmic}
\end{algorithm}

\section{Model III: AND-Tree Optimization}
\subsection{Motivation and difference from OR-tree}
An OR-tree requires \emph{one} successful alternative. An AND-tree requires the chosen division's subproblems to \emph{all} be solved and mutually compatible. For optimization, different divisions must also be compared to find the best result. The AND-tree may backtrack to another division if the current one fails or cannot beat an existing solution.

\subsection{Search model: $A_{\wedge}=(S_{\wedge},T_{\wedge})$}
\paragraph{Problem descriptions.}
We propose two kinds of problem description:
\begin{equation}
\boxed{\begin{aligned}
\Prob_{\wedge}=\;&\{\Schedule(a,U)\}\ \cup\ \{\Verify(a,x,s)\}.
\end{aligned}}
\end{equation}
Here $\Schedule(a,U)$ asks for a completion of partial assignment $a$ on the remaining components $U$. The problem $\Verify(a,x,s)$ asks whether putting component $x$ in slot $s$ is consistent with the current timetable $a$.
\paragraph{Division relation.}
Select $x\in U$ and a candidate $s\in D(x)$. One possible division is
\begin{equation}
\boxed{\begin{aligned}
\Div\big(&\Schedule(a,U),\\
&\Verify(a,x,s),\\
&\Schedule(a\cup\{x\mapsto s\},U\setminus\{x\})\big).
\end{aligned}}
\end{equation}
The division has \emph{two necessary subproblems}: check the new assignment, and schedule everything remaining with the new assignment fixed. A different candidate $s$ produces a different possible $\Div$ choice. Thus this is an AND division, not an OR-tree $\Altern$ list.
\paragraph{Solved / failed subproblems.}
\begin{align}
\Verify(a,x,s)\text{ is solved}&\iff\Consistent(a\cup\{x\mapsto s\}),\\
\Schedule(a,\varnothing)\text{ is solved}&\iff\Const(a)=\mathsf{true}.
\end{align}
If verification fails, that division cannot succeed. The controller must try a different division by the lecture's AND-tree backtracking mechanism.
\paragraph{States and transitions.}
Lecture 09 defines
\begin{equation}
\boxed{S_{\wedge}\subseteq\Atree,\quad sol\in\{\yes,\und\}.}
\end{equation}
An AND-tree state is an entire tree of subproblems, not a single partial assignment. Its transitions follow the lecture's expansion and contraction relations:
\begin{equation}
  \boxed{T_{\wedge}=\{(q,q'):\Erw_{\wedge}(q,q')\text{ or }\Erw_{\wedge}^{*}(q,q')\}.}
\end{equation}

\subsection{Search process: $P_{\wedge}=(A_{\wedge},\Env_{\wedge},K_{\wedge})$}
We propose MRV for selecting the next component and branch-and-bound for choosing/pruning divisions. The environment records the best complete timetable found so far, $a_{\Best}$, and its score
\begin{equation}
  \boxed{B=\Eval(a_{\Best}).}
\end{equation}
The valid timetable returned by OR-tree (or improved by set-based search) provides the initial incumbent $a_{\Best}$.
For a partial timetable $a$, let $\LB(a,U)$ be a sound lower bound on the penalty of \emph{every complete valid extension} of $a$. Then
\begin{equation}
\boxed{\LB(a,U)\ge B\quad\Longrightarrow\quad\text{prune this division}.}
\end{equation}
The lower bound must satisfy
\begin{equation}
  \boxed{\LB(a,U)\le\Eval(\widehat a)}
\end{equation}
for every feasible completion $\widehat a$ of $a$.

\subsection{Algorithm 3: AND-tree branch-and-bound}
\begin{algorithm}[H]
\caption{AND-Tree Optimization (control implementing $\Div$)}
\begin{algorithmic}[1]
\Require Original CSP, a complete valid incumbent $a_{\Best}$
\Ensure Best valid schedule found; optimal if exhaustive
\State $B\gets\Eval(a_{\Best})$
\Procedure{Optimize}{$a,U$}
  \If{$\LB(a,U)\ge B$}
    \State \Return \Comment{Soundly pruned division}
  \EndIf
  \If{$U=\varnothing$}
    \If{$\Const(a)$ \textbf{and} $\Eval(a)<B$}
      \State $a_{\Best}\gets a$; $B\gets\Eval(a)$
    \EndIf
    \State \Return
  \EndIf
  \State $x\gets\arg\min_{y\in U}|D_a(y)|$ \Comment{MRV}
  \ForAll{$s\in D(x)$}
    \State Consider division $\Div(\Schedule(a,U),$
    \Statex \hspace{23mm}$\Verify(a,x,s),\Schedule(a\cup\{x\mapsto s\},U\setminus\{x\}))$
    \If{$\Consistent(a\cup\{x\mapsto s\})$} \Comment{Verify child}
      \State \Call{Optimize}{$a\cup\{x\mapsto s\},U\setminus\{x\}$} \Comment{Remaining child}
    \EndIf
    \State \Comment{If needed, try alternate $\Div$ (AND-tree backtracking)}
  \EndFor
\EndProcedure
\State \Call{Optimize}{$\varnothing,X$}
\State \Return $a_{\Best}$ and completion/optimality status
\end{algorithmic}
\end{algorithm}
This is operational pseudocode for the branch-and-bound control. The formal model remains an AND-tree with $\Div$ and $\Erw_{\wedge}^{*}$; the recursive procedure implements selecting divisions and exploring them. It is not an OR-tree $\Altern$ relation.

\caption{Combined Three-Paradigm Pipeline}
\begin{algorithmic}[1]
